const origin = new URL(widget.url).origin;
const frame  = document.getElementById("widget");

frame.src = widget.url;

window.addEventListener("message", event => {

    if (event.origin !== origin) {
        return;
    }

    const { type, payload } = event.data || {};

    if (type === "octo.widget.change") {
        render(payload);                  // {uuid, unitItems, displayPricing}; repeats, full state
    }

    if (type === "octo.widget.complete") {
        submitOrder(payload);             // {uuid, unitItems, displayPricing}; once
    }

    if (type === "octo.widget.resize") {
        frame.style.height = payload.height + "px";   // {uuid, height}
    }
});
